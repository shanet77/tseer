# Quadlet pod: tseer + redis share one network namespace, so tseer reaches
# redis at localhost:6379 and redis stays unpublished to the host.
# Install alongside tseer.container and redis.container in
#   /etc/containers/systemd/           (rootful)
#   ~/.config/containers/systemd/      (rootless)
# The pod generates tseer-pod.service; starting it brings up both containers.

[Unit]
Description=tseer pod (web + redis cache)

[Pod]
PodName=tseer
PublishPort=8080:8080

[Install]
WantedBy=default.target
